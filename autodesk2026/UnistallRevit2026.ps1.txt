<#
.SYNOPSIS
    Desinstalacion silenciosa de Autodesk Revit 2026 y sus componentes para Deep Freeze Cloud.

.DESCRIPTION
    1. Cierra procesos activos de Revit para evitar bloqueos de archivos.
    2. Busca en el Registro de Windows (HKLM) todas las entradas de "Revit 2026".
    3. Prioriza la desinstalacion del producto principal y luego sus complementos/idiomas
       usando ODIS (Installer.exe) o msiexec en modo silencioso.
    4. Registra todo el proceso en C:\LogsDFC\Uninstall-Revit2026.log.

    IMPORTANTE: Si el equipo tiene Deep Freeze activo en estado Frozen, los cambios
    se revertiran al reiniciar. Ejecutar en estado Thawed.

.NOTES
    Codigos de salida:
      0 = Desinstalacion completada o Revit 2026 no estaba instalado
      1 = Error critico o algun componente fallo al desinstalar
#>
[CmdletBinding()]
param(
    [string]$LogFolder = 'C:\LogsDFC'
)

$ErrorActionPreference = 'Stop'
$LogFile = Join-Path $LogFolder 'Uninstall-Revit2026.log'

if (-not (Test-Path -LiteralPath $LogFolder)) {
    New-Item -ItemType Directory -Path $LogFolder -Force | Out-Null
}

function Write-Log {
    param([string]$Message, [string]$Level = 'INFO')
    $line = '{0} [{1}] {2}' -f (Get-Date -Format 'yyyy-MM-dd HH:mm:ss'), $Level, $Message
    Write-Output $line
    try { Add-Content -LiteralPath $LogFile -Value $line } catch { }
}

Write-Log '=== Iniciando proceso de desinstalacion de Autodesk Revit 2026 ==='

# 1. Cerrar procesos activos relacionados con Revit
$procesos = @('Revit', 'RevitAccelerator', 'AdskAccessCore', 'AdskAccessServiceHost')
foreach ($proc in $procesos) {
    Get-Process -Name $proc -ErrorAction SilentlyContinue | ForEach-Object {
        Write-Log "Deteniendo proceso activo: $($_.Name) (PID: $($_.Id))"
        Stop-Process -Id $_.Id -Force -ErrorAction SilentlyContinue
    }
}

# 2. Buscar entradas de Revit 2026 en el Registro
$regPaths = @(
    'HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*',
    'HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*'
)

$entradas = Get-ItemProperty -Path $regPaths -ErrorAction SilentlyContinue | Where-Object {
    $_.DisplayName -match 'Revit\s*2026'
}

if (-not $entradas) {
    Write-Log 'No se encontro Autodesk Revit 2026 instalado en este equipo.'
    exit 0
}

# Ordenar: primero el paquete principal ("Autodesk Revit 2026") y despues idiomas/complementos
$entradasOrdenadas = $entradas | Sort-Object @{
    Expression = { if ($_.DisplayName -match '^Autodesk Revit 2026(\s|$)') { 0 } else { 1 } }
}, DisplayName

$huboErrores = $false

foreach ($app in $entradasOrdenadas) {
    $nombre = $app.DisplayName
    $quietCmd = $app.QuietUninstallString
    $stdCmd   = $app.UninstallString

    Write-Log "Procesando: $nombre"

    try {
        if ($quietCmd) {
            # Caso 1: Ya incluye cadena silenciosa (comun en Autodesk ODIS: Installer.exe -i uninstall -q ...)
            Write-Log "Ejecutando QuietUninstallString: $quietCmd"
            $proc = Start-Process -FilePath 'cmd.exe' -ArgumentList "/c `"$quietCmd`"" -Wait -PassThru -WindowStyle Hidden
        }
        elseif ($stdCmd -match 'Installer\.exe') {
            # Caso 2: Autodesk ODIS sin bandera -q explicita
            $cmdOdis = "$stdCmd -q"
            Write-Log "Ejecutando ODIS silencioso: $cmdOdis"
            $proc = Start-Process -FilePath 'cmd.exe' -ArgumentList "/c `"$cmdOdis`"" -Wait -PassThru -WindowStyle Hidden
        }
        elseif ($stdCmd -match 'msiexec\.exe') {
            # Caso 3: Instalador MSI tradicional -> extraer GUID y forzar /x /qn
            if ($stdCmd -match '\{[0-9A-Fa-f\-]{36}\}') {
                $guid = $Matches[0]
                Write-Log "Ejecutando msiexec /x $guid /qn /norestart"
                $proc = Start-Process -FilePath 'msiexec.exe' -ArgumentList "/x $guid /qn /norestart" -Wait -PassThru -WindowStyle Hidden
            } else {
                Write-Log "No se pudo extraer GUID de: $stdCmd" 'WARN'
                continue
            }
        }
        else {
            Write-Log "Cadena de desinstalacion no reconocida para '$nombre': $stdCmd" 'WARN'
            continue
        }

        $code = $proc.ExitCode
        # 0 = Exito | 3010 = Exito, requiere reinicio | 1605 = Ya estaba desinstalado
        if ($code -in @(0, 3010, 1605)) {
            Write-Log "Desinstalado correctamente '$nombre' (Codigo de salida: $code)."
        } else {
            Write-Log "Advertencia/Error al desinstalar '$nombre'. Codigo de salida: $code" 'ERROR'
            $huboErrores = $true
        }
    }
    catch {
        Write-Log "Excepcion al desinstalar '$nombre': $($_.Exception.Message)" 'ERROR'
        $huboErrores = $true
    }
}

if ($huboErrores) {
    Write-Log 'Proceso finalizado con errores en uno o mas componentes. Revise el log.' 'ERROR'
    exit 1
} else {
    Write-Log '=== Desinstalacion de Autodesk Revit 2026 finalizada con exito ==='
    exit 0
}
